-- backfilled while ALTER TABLE holds its transactional lock.

CREATE FUNCTION app_private.origin_revision_trigger() RETURNS trigger
LANGUAGE plpgsql SET search_path='' AS $fn$
BEGIN
 IF NEW.origin IS NULL THEN NEW.origin_revision:=NULL;
 ELSE NEW.origin_revision:=pg_catalog.gen_random_uuid(); END IF;
 RETURN NEW;
END $fn$;
REVOKE ALL ON FUNCTION app_private.origin_revision_trigger() FROM PUBLIC,anon,authenticated,service_role;

LANGUAGE plpgsql SET search_path='' AS $fn$

END $fn$;

-- The lock order is identical to draft_withdraw_current. No UTM leaves this RPC.
CREATE FUNCTION app_private.origin_withdrawal_current_v2(p_user uuid,p_session uuid) RETURNS jsonb
LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $fn$
DECLARE n integer; revision uuid;
BEGIN
 IF p_user IS NULL OR p_session IS NULL THEN RAISE EXCEPTION 'Unavailable.' USING ERRCODE='42501'; END IF;
 PERFORM 1 FROM auth.users WHERE id=p_user FOR SHARE;
 PERFORM 1 FROM auth.sessions WHERE id=p_session AND user_id=p_user FOR SHARE;
 PERFORM 1 FROM public.user_moderation WHERE user_id=p_user FOR SHARE;
 IF app_private.draft_withdraw_identity(p_user,p_session) IS NOT TRUE THEN RAISE EXCEPTION 'Unavailable.' USING ERRCODE='42501'; END IF;
 PERFORM pg_catalog.pg_advisory_xact_lock(pg_catalog.hashtextextended(p_user::text || ':signup',0));
 PERFORM 1 FROM app_private.signup_intents WHERE user_id=p_user ORDER BY id FOR UPDATE;
 PERFORM 1 FROM app_private.user_admission WHERE user_id=p_user FOR UPDATE;
 PERFORM 1 FROM app_private.signup_completion_fences_draft WHERE user_id=p_user FOR UPDATE;
 PERFORM 1 FROM app_private.signup_origins WHERE user_id=p_user FOR UPDATE;
 SELECT count(*)::integer INTO n FROM (
  SELECT origin_revision FROM app_private.signup_intents WHERE user_id=p_user AND state='bound' AND origin IS NOT NULL AND expires_at>pg_catalog.clock_timestamp()
  UNION ALL SELECT origin_revision FROM app_private.signup_origins WHERE user_id=p_user AND expires_at>pg_catalog.clock_timestamp()
 ) AS origins;
 IF n>1 THEN RAISE EXCEPTION 'Unavailable.' USING ERRCODE='42501'; END IF;
 IF n=0 THEN RETURN pg_catalog.jsonb_build_object('status','absent','revision',NULL); END IF;
 SELECT origin_revision INTO STRICT revision FROM (
  SELECT origin_revision FROM app_private.signup_intents WHERE user_id=p_user AND state='bound' AND origin IS NOT NULL AND expires_at>pg_catalog.clock_timestamp()
  UNION ALL SELECT origin_revision FROM app_private.signup_origins WHERE user_id=p_user AND expires_at>pg_catalog.clock_timestamp()
 ) AS origins;
 RETURN pg_catalog.jsonb_build_object('status','active','revision',revision);
END $fn$;

CREATE FUNCTION public.signup_origin_status_v2(p_user uuid,p_session uuid) RETURNS jsonb
LANGUAGE sql SECURITY DEFINER SET search_path='' AS $fn$
 SELECT app_private.origin_withdrawal_current_v2(p_user,p_session)
$fn$;
CREATE FUNCTION public.signup_origin_withdraw_v2(p_user uuid,p_session uuid,p_revision uuid) RETURNS jsonb
LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $fn$
DECLARE current_state jsonb; outcome jsonb;
BEGIN
 IF p_revision IS NULL THEN RAISE EXCEPTION 'Unavailable.' USING ERRCODE='42501'; END IF;
 current_state:=app_private.origin_withdrawal_current_v2(p_user,p_session);
 IF current_state->>'status'='absent' THEN RETURN pg_catalog.jsonb_build_object('status','absent'); END IF;
 IF current_state->>'revision' IS DISTINCT FROM p_revision::text THEN RETURN pg_catalog.jsonb_build_object('status','stale'); END IF;
 outcome:=app_private.draft_withdraw_current(p_user,p_session);
 IF outcome IS DISTINCT FROM pg_catalog.jsonb_build_object('status','withdrawn') AND outcome IS DISTINCT FROM pg_catalog.jsonb_build_object('status','absent') THEN RAISE EXCEPTION 'Unavailable.' USING ERRCODE='42501'; END IF;
 RETURN outcome;
END $fn$;
REVOKE ALL ON FUNCTION app_private.origin_withdrawal_current_v2(uuid,uuid),public.signup_origin_status_v2(uuid,uuid),public.signup_origin_withdraw_v2(uuid,uuid,uuid) FROM PUBLIC,anon,authenticated,service_role;
GRANT EXECUTE ON FUNCTION public.signup_origin_status_v2(uuid,uuid),public.signup_origin_withdraw_v2(uuid,uuid,uuid) TO service_role;
