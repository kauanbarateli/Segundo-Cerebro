-- SOMENTE LEITURA: metadados e contagens agregadas; não consulta linhas de usuários.
-- O projeto deve ser confirmado independentemente pelo painel/MCP antes da execução.
-- Um snapshot não bloqueia conexões, DDL ou novas migrations e não autoriza instalação.
SELECT jsonb_build_object(
  'database', current_database(),
  'server_version', current_setting('server_version_num'),
  'executor', current_user,
  'observed_at', clock_timestamp(),
  'feedback_objects', jsonb_build_object(
    'requests_exists', to_regclass('app_private.feedback_requests') IS NOT NULL,
    'barriers_exist', to_regclass('app_private.feedback_barriers_draft') IS NOT NULL,
    'functions', (SELECT coalesce(jsonb_agg(jsonb_build_object(
      'schema', n.nspname, 'name', p.proname,
      'arguments', pg_get_function_identity_arguments(p.oid)
    ) ORDER BY n.nspname,p.proname,p.oid), '[]'::jsonb)
    FROM pg_proc p JOIN pg_namespace n ON n.oid=p.pronamespace
    WHERE n.nspname IN ('app_private','public') AND p.proname IN
      ('feedback_list','feedback_submit','feedback_barrier_submit','feedback_barrier_status','feedback_barrier_cancel',
       'draft_feedback_lock','draft_feedback_actor','draft_feedback_submit','draft_feedback_status','draft_feedback_cancel'))
  ),
  'dependencies', (SELECT coalesce(jsonb_agg(jsonb_build_object(
      'name', p.proname, 'arguments', pg_get_function_identity_arguments(p.oid),
      'definition_sha256', encode(sha256(convert_to(pg_get_functiondef(p.oid),'UTF8')),'hex'),
      'owner', pg_get_userbyid(p.proowner), 'security_definer', p.prosecdef,
      'config_sha256', encode(sha256(convert_to(coalesce(array_to_string(p.proconfig,E'\n'),''),'UTF8')),'hex'),
      'language', l.lanname,
      'anon_execute', has_function_privilege('anon',p.oid,'EXECUTE'),
      'authenticated_execute', has_function_privilege('authenticated',p.oid,'EXECUTE'),
      'service_role_execute', has_function_privilege('service_role',p.oid,'EXECUTE')
    ) ORDER BY p.proname,p.oid), '[]'::jsonb)
    FROM pg_proc p JOIN pg_namespace n ON n.oid=p.pronamespace JOIN pg_language l ON l.oid=p.prolang
    WHERE n.nspname='app_private' AND p.proname IN ('require_actor','session_active','append_event')),
  'required_tables', (SELECT coalesce(jsonb_agg(jsonb_build_object(
      'schema', n.nspname, 'table', c.relname, 'kind', c.relkind,
      'owner', pg_get_userbyid(c.relowner), 'rls', c.relrowsecurity, 'force_rls', c.relforcerowsecurity
    ) ORDER BY n.nspname,c.relname), '[]'::jsonb)
    FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace
    WHERE (n.nspname,c.relname) IN (('auth','users'),('auth','sessions'),('public','user_moderation'),('public','domain_events'))),
  'app_private_schema_usage', jsonb_build_object(
    'anon', has_schema_privilege('anon','app_private','USAGE'),
    'authenticated', has_schema_privilege('authenticated','app_private','USAGE'),
    'service_role', has_schema_privilege('service_role','app_private','USAGE')),
  'event_table', (SELECT jsonb_build_object(
      'total_bytes', pg_total_relation_size(c.oid), 'estimated_rows', c.reltuples,
      'owner', pg_get_userbyid(c.relowner), 'rls', c.relrowsecurity, 'force_rls', c.relforcerowsecurity,
      'constraint', (SELECT jsonb_build_object(
        'definition_sha256', encode(sha256(convert_to(pg_get_constraintdef(k.oid),'UTF8')),'hex'),
        'expression_sha256', encode(sha256(convert_to(pg_get_expr(k.conbin,k.conrelid),'UTF8')),'hex'),
        'type', k.contype, 'validated', k.convalidated, 'deferrable', k.condeferrable,
        'deferred', k.condeferred, 'noinherit', k.connoinherit)
        FROM pg_constraint k WHERE k.conrelid=c.oid AND k.conname='domain_events_entity_type_check'),
      'user_fk', (SELECT coalesce(jsonb_agg(jsonb_build_object('name',k.conname,'definition',pg_get_constraintdef(k.oid)) ORDER BY k.conname),'[]'::jsonb)
        FROM pg_constraint k WHERE k.conrelid=c.oid AND k.contype='f'),
      'user_triggers', (SELECT coalesce(jsonb_agg(jsonb_build_object('name',t.tgname,'enabled',t.tgenabled) ORDER BY t.tgname),'[]'::jsonb)
        FROM pg_trigger t WHERE t.tgrelid=c.oid AND NOT t.tgisinternal),
      'locks', (SELECT coalesce(jsonb_agg(jsonb_build_object('mode',x.mode,'granted',x.granted,'count',x.n) ORDER BY x.mode,x.granted),'[]'::jsonb)
        FROM (SELECT mode,granted,count(*) AS n FROM pg_locks WHERE relation=c.oid GROUP BY mode,granted) x)
    ) FROM pg_class c WHERE c.oid=to_regclass('public.domain_events')),
  'ledger', (SELECT jsonb_build_object(
      'count',count(*),
      'fingerprint_sha256',encode(sha256(convert_to(coalesce(string_agg(version||':'||coalesce(name,''),E'\n' ORDER BY version,name),''),'UTF8')),'hex'),
      'recent', (SELECT coalesce(jsonb_agg(jsonb_build_object('version',x.version,'name',x.name) ORDER BY x.version DESC),'[]'::jsonb)
        FROM (SELECT version,name FROM supabase_migrations.schema_migrations ORDER BY version DESC LIMIT 10) x))
    FROM supabase_migrations.schema_migrations),
  'activity', (SELECT jsonb_build_object(
      'active_sessions',count(*) FILTER (WHERE state='active' AND pid<>pg_backend_pid()),
      'open_transactions',count(*) FILTER (WHERE xact_start IS NOT NULL AND pid<>pg_backend_pid()))
    FROM pg_stat_activity WHERE datname=current_database())
)::text AS resultado;
