-- The binding pointer is an opaque random secret, stored here only as SHA-256.
-- The subject is a server-created HMAC, never an email or a client authority.
create function app_private.valid_signup_origin(p_origin jsonb) returns boolean
language sql immutable set search_path='' as $$
 select coalesce(jsonb_typeof(p_origin)='object' and p_origin<>'{}'::jsonb
  and octet_length(p_origin::text)<=1024
  and p_origin-array['utm_source','utm_medium','utm_campaign','utm_term','utm_content']='{}'::jsonb
  and not exists(select 1 from jsonb_each(p_origin) e where jsonb_typeof(e.value)<>'string'
   or char_length(e.value #>> '{}') not between 1 and 200 or e.value #>> '{}' <> btrim(e.value #>> '{}')
   or (e.value #>> '{}') ~ '[[:cntrl:]<>]'),false);
$$;
revoke all on function app_private.valid_signup_origin(jsonb) from public,anon,authenticated,service_role;

    or (app_private.valid_signup_origin(origin) and origin_consent_version is not null and char_length(origin_consent_version) between 1 and 100 and origin_consented_at is not null)),
 check(state<>'consumed' or origin is null),
 check((state='pending' and user_id is null and binding_hash is not null and subject_hash is not null and consumed_at is null)
    or (state='bound' and user_id is not null and binding_hash is null and subject_hash is not null and consumed_at is null)
    or (state='consumed' and user_id is not null and binding_hash is null and consumed_at is not null))

create function app_private.signup_intent_create(p_binding_hash text,p_subject_hash text,p_terms text,p_privacy text,p_expires timestamptz,p_origin jsonb,p_origin_version text) returns uuid
language plpgsql security definer set search_path='' as $$ declare v_id uuid; begin
 perform 1 from app_private.signup_settings where singleton and enabled for share;
 if not found then raise exception 'Signup unavailable.' using errcode='42501'; end if;
 if p_binding_hash is null or p_binding_hash !~ '^[a-f0-9]{64}$' or p_subject_hash is null or p_subject_hash !~ '^[a-f0-9]{64}$'
  or p_terms is null or char_length(p_terms) not between 1 and 100 or p_privacy is null or char_length(p_privacy) not between 1 and 100
  or p_expires is null or p_expires<=now() or p_expires>now()+interval '1 hour'
  or (p_origin is null and p_origin_version is not null)
  or (p_origin is not null and (not app_private.valid_signup_origin(p_origin) or p_origin_version is null or char_length(p_origin_version) not between 1 and 100)) then
  raise exception 'Invalid signup intent.' using errcode='22023';
 end if;
 insert into app_private.signup_intents(binding_hash,subject_hash,terms_version,privacy_version,expires_at,origin,origin_consent_version,origin_consented_at)
 values(p_binding_hash,p_subject_hash,p_terms,p_privacy,p_expires,p_origin,p_origin_version,case when p_origin is null then null else now() end) returning id into v_id;
 return v_id;
end $$;

-- For pending users only: validate session and moderation without granting
-- module access. No user-controlled metadata participates in this predicate.
create function app_private.signup_pending_identity(p_user uuid,p_session uuid) returns boolean
language sql stable security definer set search_path='' as $$
 select p_user is not null and p_session is not null and exists(
  select 1 from auth.users u join auth.sessions s on s.user_id=u.id
  join public.user_moderation m on m.user_id=u.id
  where u.id=p_user and s.id=p_session and u.email_confirmed_at is not null
   and not u.is_anonymous and u.deleted_at is null and (u.banned_until is null or u.banned_until<=now())
   and (s.not_after is null or s.not_after>now()) and m.status='active' and not m.must_change_password
 );
$$;
revoke all on function app_private.signup_pending_identity(uuid,uuid) from public,anon,authenticated,service_role;

-- After OTP verification, the server can find only the intent bound to the
-- verified Auth ID. The subject hash must be recomputed from verified identity.
create function app_private.signup_find_bound_intent(p_user uuid,p_session uuid,p_subject_hash text) returns jsonb
language plpgsql security definer set search_path='' as $$ declare i app_private.signup_intents; begin
 if not app_private.signup_pending_identity(p_user,p_session) then raise exception 'Signup unavailable.' using errcode='42501'; end if;
 select * into i from app_private.signup_intents where user_id=p_user and subject_hash=p_subject_hash and
  ((state='bound' and expires_at>now()) or (state='consumed' and exists(select 1 from app_private.user_admission a where a.user_id=p_user and a.origin='public' and a.state='admitted')));
 if not found then return null; end if;
 if i.state='bound' and not exists(select 1 from app_private.signup_settings where singleton and enabled) then raise exception 'Signup unavailable.' using errcode='42501'; end if;
 return jsonb_build_object('intentId',i.id,'termsVersion',i.terms_version,'privacyVersion',i.privacy_version,'state',i.state,'originPending',i.origin is not null);
end $$;

-- Only the server path that has just verified a signup OTP may call this.
-- This recovers an expired/missing intent for an existing pending identity;
-- ownership of the email alone does not admit the identity or keep its password.
create function app_private.signup_recover_verified_intent(p_user uuid,p_session uuid,p_subject_hash text,p_terms text,p_privacy text) returns jsonb
language plpgsql security definer set search_path='' as $$ declare i app_private.signup_intents; a app_private.user_admission; begin
 perform 1 from auth.users where id=p_user for share;
 perform 1 from auth.sessions where id=p_session and user_id=p_user for share;
 perform 1 from public.user_moderation where user_id=p_user for share;
 if not app_private.signup_pending_identity(p_user,p_session) then raise exception 'Signup unavailable.' using errcode='42501'; end if;
 perform pg_catalog.pg_advisory_xact_lock(pg_catalog.hashtextextended(p_user::text || ':signup',0));
 perform 1 from app_private.signup_settings where singleton and enabled for share;
 if not found or p_subject_hash is null or p_subject_hash !~ '^[a-f0-9]{64}$' or p_terms is null or char_length(p_terms) not between 1 and 100 or p_privacy is null or char_length(p_privacy) not between 1 and 100
  then raise exception 'Signup unavailable.' using errcode='42501'; end if;
 -- Under the same advisory lock, never replace an intent with an extant fence.
 if pg_catalog.to_regclass('app_private.signup_completion_fences_draft') is null then raise exception 'Signup unavailable.' using errcode='42501'; end if;
 if exists(select 1 from app_private.signup_completion_fences_draft f where f.user_id=p_user) then raise exception 'Signup unavailable.' using errcode='42501'; end if;
 select * into a from app_private.user_admission where user_id=p_user for update;
 if not found or a.origin<>'unclassified' or a.state<>'pending' then raise exception 'Signup unavailable.' using errcode='42501'; end if;
 select * into i from app_private.signup_intents where user_id=p_user for update;
 if found then
  if exists(select 1 from app_private.signup_completion_fences_draft f where f.intent_id=i.id) then raise exception 'Signup unavailable.' using errcode='42501'; end if;
  if i.state<>'bound' or i.subject_hash is distinct from p_subject_hash then raise exception 'Signup unavailable.' using errcode='42501'; end if;
  if i.expires_at<=pg_catalog.clock_timestamp() then delete from app_private.signup_intents where id=i.id; i:=null;
  else update app_private.signup_intents set terms_version=p_terms,privacy_version=p_privacy where id=i.id returning * into i; end if;
 end if;
 if i.id is null then
  insert into app_private.signup_intents(binding_hash,subject_hash,terms_version,privacy_version,expires_at,state,user_id)
   values(null,p_subject_hash,p_terms,p_privacy,pg_catalog.clock_timestamp()+interval '30 minutes','bound',p_user) returning * into i;
 end if;
 return jsonb_build_object('intentId',i.id,'termsVersion',i.terms_version,'privacyVersion',i.privacy_version,'state',i.state,'originPending',i.origin is not null);
end $$;

create function app_private.signup_complete(p_user uuid,p_session uuid,p_intent uuid,p_subject_hash text,p_terms text,p_privacy text,p_accept_origin boolean,p_origin_version text) returns jsonb
language plpgsql security definer set search_path='' as $$ declare i app_private.signup_intents; a app_private.user_admission; v_enabled boolean; begin
 perform 1 from auth.users where id=p_user for share;
 perform 1 from auth.sessions where id=p_session and user_id=p_user for share;
 perform 1 from public.user_moderation where user_id=p_user for share;
 if not app_private.signup_pending_identity(p_user,p_session) then raise exception 'Signup unavailable.' using errcode='42501'; end if;
 perform pg_catalog.pg_advisory_xact_lock(pg_catalog.hashtextextended(p_user::text || ':signup',0));
 select enabled into v_enabled from app_private.signup_settings where singleton for share;
 select * into i from app_private.signup_intents where id=p_intent for update;
 if not found or i.user_id is distinct from p_user or i.subject_hash is distinct from p_subject_hash or i.terms_version is distinct from p_terms or i.privacy_version is distinct from p_privacy
  then raise exception 'Signup unavailable.' using errcode='42501'; end if;
 select * into a from app_private.user_admission where user_id=p_user for update;
 if not found then raise exception 'Signup unavailable.' using errcode='42501'; end if;

-- Operator-invoked, bounded retention; no schedule is installed by this SQL.
CREATE FUNCTION app_private.signup_prune() RETURNS jsonb
LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $$
DECLARE r record; locked_id uuid; cutoff timestamptz;
 removed integer:=0; scrubbed integer:=0; origins_removed integer:=0; protected integer:=0; expired_origins integer:=0; n integer;
BEGIN
  IF pg_catalog.current_setting('transaction_isolation') IS DISTINCT FROM 'read committed' THEN
    RAISE EXCEPTION 'signup_prune requires read committed';
  END IF;
 cutoff:=pg_catalog.clock_timestamp();
 -- Read candidates without row locks, then take the per-user advisory lock
 -- before locking bound rows. Try/skip avoids waiting on competing holders.
 FOR r IN SELECT i.id,i.user_id,i.state FROM app_private.signup_intents i
  WHERE i.state IN ('pending','bound') AND i.expires_at<=cutoff
   AND (i.origin IS NOT NULL OR NOT EXISTS(SELECT 1 FROM app_private.signup_completion_fences_draft f WHERE f.intent_id=i.id OR (i.user_id IS NOT NULL AND f.user_id=i.user_id)))
  ORDER BY i.expires_at,i.id LIMIT 500 LOOP
  IF r.state='bound' THEN
   IF r.user_id IS NULL OR NOT pg_catalog.pg_try_advisory_xact_lock(pg_catalog.hashtextextended(r.user_id::text || ':signup',0)) THEN CONTINUE; END IF;
   SELECT i.id INTO locked_id FROM app_private.signup_intents i WHERE i.id=r.id AND i.user_id=r.user_id AND i.state='bound' AND i.expires_at<=pg_catalog.clock_timestamp() FOR UPDATE SKIP LOCKED;
   IF NOT FOUND THEN CONTINUE; END IF;
   -- Separate statement after lock acquisition; every new writer must obey
   -- the same advisory lock. Old or owner-level bypass remains a cutover gate.
   IF EXISTS(SELECT 1 FROM app_private.signup_completion_fences_draft WHERE user_id=r.user_id OR intent_id=r.id) THEN
    protected:=protected+1;
    -- Even when existing identifiers disagree, remove expired attribution
    -- from this locked intent; never change a mismatched fence.
    UPDATE app_private.signup_intents SET origin=NULL,origin_consent_version=NULL,origin_consented_at=NULL WHERE id=r.id AND origin IS NOT NULL;
    GET DIAGNOSTICS n=ROW_COUNT; expired_origins:=expired_origins+n;
    UPDATE app_private.signup_completion_fences_draft SET origin_expired=true,updated_at=pg_catalog.now() WHERE user_id=r.user_id AND intent_id=r.id AND NOT origin_expired;
    CONTINUE;
   END IF;
   DELETE FROM app_private.signup_intents WHERE id=r.id AND user_id=r.user_id AND state='bound' AND expires_at<=pg_catalog.clock_timestamp()
    AND NOT EXISTS(SELECT 1 FROM app_private.signup_completion_fences_draft WHERE user_id=r.user_id OR intent_id=r.id);
  ELSE
   -- Unbound intent: lock row directly; it cannot have a normal fence until
   -- binding, which also needs that row. Recheck state and absence after lock.
   SELECT i.id INTO locked_id FROM app_private.signup_intents i WHERE i.id=r.id AND i.user_id IS NULL AND i.state='pending' AND i.expires_at<=pg_catalog.clock_timestamp() FOR UPDATE SKIP LOCKED;
   IF NOT FOUND THEN CONTINUE; END IF;
   DELETE FROM app_private.signup_intents WHERE id=r.id AND user_id IS NULL AND state='pending' AND expires_at<=pg_catalog.clock_timestamp()
    AND NOT EXISTS(SELECT 1 FROM app_private.signup_completion_fences_draft WHERE intent_id=r.id);
  END IF;
  GET DIAGNOSTICS n=ROW_COUNT; removed:=removed+n;
 END LOOP;
 UPDATE app_private.signup_intents SET subject_hash=NULL WHERE id IN (
  SELECT id FROM app_private.signup_intents WHERE state='consumed' AND subject_hash IS NOT NULL AND consumed_at<=cutoff-interval '24 hours'
  ORDER BY consumed_at,id FOR UPDATE SKIP LOCKED LIMIT 500
 );
 GET DIAGNOSTICS scrubbed=ROW_COUNT;
 DELETE FROM app_private.signup_origins WHERE user_id IN (
  SELECT user_id FROM app_private.signup_origins WHERE expires_at<=cutoff ORDER BY expires_at,user_id FOR UPDATE SKIP LOCKED LIMIT 500
 );
 GET DIAGNOSTICS origins_removed=ROW_COUNT;
 RETURN pg_catalog.jsonb_build_object('expiredRemoved',removed,'consumedScrubbed',scrubbed,'originsRemoved',origins_removed,'fencedExamined',protected,'expiredOriginsCleared',expired_origins);
END $$;
create function public.signup_prune() returns jsonb language sql security invoker set search_path='' as $$ select app_private.signup_prune() $$;
revoke all on function app_private.signup_prune(),public.signup_prune() from public,anon,authenticated,service_role;
grant execute on function app_private.signup_prune(),public.signup_prune() to service_role;

insert into app_private.user_admission(user_id,origin,state,admitted_at)

create function app_private.admission_new_identity() returns trigger
language plpgsql security definer set search_path='' as $$ declare pointer text; begin
 insert into app_private.user_admission(user_id,origin,state) values(new.id,'unclassified','pending');

 -- subsequent metadata updates or a duplicate signup cannot bind an old account.
 pointer:=new.raw_user_meta_data->>'sc_signup_pointer';
 if pointer is not null and pointer ~ '^[A-Za-z0-9_-]{43,128}$' then
  update app_private.signup_intents set state='bound',user_id=new.id,binding_hash=null
   where binding_hash=encode(sha256(convert_to(pointer,'UTF8')),'hex') and state='pending' and expires_at>now();

 -- verify GoTrue does not subsequently restore it before using this candidate.
 if new.raw_user_meta_data ? 'sc_signup_pointer' then
  update auth.users set raw_user_meta_data=raw_user_meta_data-'sc_signup_pointer' where id=new.id;
 end if;
 return new;
end $$;

 );
$$;

CREATE FUNCTION app_private.draft_completion_context(p_user uuid,p_session uuid,p_intent uuid,p_subject_hash text,p_require_enabled boolean,p_require_live boolean)
RETURNS void LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $$
DECLARE v_intent app_private.signup_intents; v_admission app_private.user_admission;
BEGIN
 IF p_user IS NULL OR p_session IS NULL OR p_intent IS NULL OR p_subject_hash IS NULL OR p_subject_hash !~ '^[a-f0-9]{64}$' OR p_require_enabled IS NULL OR p_require_live IS NULL THEN
  RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501';
 END IF;
 PERFORM 1 FROM auth.users WHERE id=p_user FOR SHARE;
 PERFORM 1 FROM auth.sessions WHERE id=p_session AND user_id=p_user FOR SHARE;
 PERFORM 1 FROM public.user_moderation WHERE user_id=p_user FOR SHARE;
 IF app_private.signup_pending_identity(p_user,p_session) IS NOT TRUE THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 PERFORM pg_catalog.pg_advisory_xact_lock(pg_catalog.hashtextextended(p_user::text || ':signup',0));
 IF p_require_enabled THEN
  PERFORM 1 FROM app_private.signup_settings WHERE singleton AND enabled FOR SHARE;
  IF NOT FOUND THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 END IF;
 SELECT * INTO v_intent FROM app_private.signup_intents WHERE id=p_intent FOR UPDATE;
 IF NOT FOUND OR v_intent.user_id IS DISTINCT FROM p_user OR v_intent.subject_hash IS DISTINCT FROM p_subject_hash
  OR v_intent.state IS DISTINCT FROM 'bound' OR v_intent.expires_at IS NULL OR (p_require_live AND v_intent.expires_at<=pg_catalog.clock_timestamp()) THEN
  RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501';
 END IF;
 SELECT * INTO v_admission FROM app_private.user_admission WHERE user_id=p_user FOR UPDATE;
 IF NOT FOUND OR v_admission.origin IS DISTINCT FROM 'unclassified' OR v_admission.state IS DISTINCT FROM 'pending' THEN
  RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501';
 END IF;
 -- The admission lock may itself wait beyond the intent deadline.
 IF p_require_live AND (v_intent.expires_at IS NULL OR v_intent.expires_at<=pg_catalog.clock_timestamp()) THEN
  RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501';
 END IF;
END $$;

CREATE FUNCTION app_private.draft_reserve_completion(p_user uuid,p_session uuid,p_intent uuid,p_subject_hash text,p_request uuid,p_terms text,p_privacy text,p_accept_origin boolean,p_origin_version text)
RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $$
DECLARE i app_private.signup_intents; v_generation uuid; v_effective boolean;
BEGIN
 IF p_request IS NULL OR p_terms IS NULL OR p_privacy IS NULL OR p_accept_origin IS NULL THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 PERFORM app_private.draft_completion_context(p_user,p_session,p_intent,p_subject_hash,true,true);
 SELECT * INTO i FROM app_private.signup_intents WHERE id=p_intent;
 IF i.terms_version IS DISTINCT FROM p_terms OR i.privacy_version IS DISTINCT FROM p_privacy THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 v_effective := p_accept_origin AND i.origin IS NOT NULL;
 IF (v_effective AND (p_origin_version IS NULL OR p_origin_version IS DISTINCT FROM i.origin_consent_version)) OR (NOT v_effective AND p_origin_version IS NOT NULL) THEN
  RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501';
 END IF;
 -- A losing request cannot alter the winning decision here. Withdrawal is a
 -- separate explicitly authorized monotonic operation below.
 IF EXISTS(SELECT 1 FROM app_private.signup_completion_fences_draft WHERE user_id=p_user) THEN RETURN pg_catalog.jsonb_build_object('status','pending'); END IF;
 IF NOT v_effective AND i.origin IS NOT NULL THEN
  UPDATE app_private.signup_intents SET origin=NULL,origin_consent_version=NULL,origin_consented_at=NULL WHERE id=p_intent;
 END IF;
 INSERT INTO app_private.signup_completion_fences_draft(user_id,intent_id,owner_session_id,request_id,generation,phase,terms_version,privacy_version,origin_accepted,origin_consent_version)
 VALUES(p_user,p_intent,p_session,p_request,pg_catalog.gen_random_uuid(),'reserved',p_terms,p_privacy,v_effective,CASE WHEN v_effective THEN p_origin_version ELSE NULL END)
 ON CONFLICT DO NOTHING RETURNING generation INTO v_generation;
 IF v_generation IS NULL THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 RETURN pg_catalog.jsonb_build_object('status','acquired','requestId',p_request,'generation',v_generation);
END $$;

-- Server assertion only: no SQL statement can prove the external GoTrue effect.
-- Identity and same session must be verified again by trusted server after Auth.
CREATE FUNCTION app_private.draft_ack_with_proof(p_user uuid,p_session uuid,p_intent uuid,p_subject_hash text,p_request uuid,p_generation uuid)
RETURNS boolean LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $$
DECLARE f app_private.signup_completion_fences_draft; i app_private.signup_intents;
BEGIN
 PERFORM app_private.draft_completion_context(p_user,p_session,p_intent,p_subject_hash,true,true);
 SELECT * INTO f FROM app_private.signup_completion_fences_draft WHERE user_id=p_user AND intent_id=p_intent
  AND owner_session_id=p_session AND request_id=p_request AND generation=p_generation FOR UPDATE;
 IF NOT FOUND OR f.phase IS DISTINCT FROM 'auth_in_flight' THEN RETURN false; END IF;
 SELECT * INTO STRICT i FROM app_private.signup_intents WHERE id=p_intent;
 INSERT INTO app_private.signup_password_proofs(intent_id,user_id,session_id,expires_at)
 VALUES(i.id,p_user,p_session,least(i.expires_at,pg_catalog.now()+interval '10 minutes'))
 ON CONFLICT(intent_id) DO UPDATE SET user_id=excluded.user_id,session_id=excluded.session_id,
  expires_at=excluded.expires_at,consumed_at=NULL
 WHERE app_private.signup_password_proofs.consumed_at IS NULL;
 IF NOT FOUND THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 UPDATE app_private.signup_completion_fences_draft SET phase='password_acknowledged',updated_at=pg_catalog.now()
  WHERE user_id=p_user AND intent_id=p_intent AND owner_session_id=p_session AND request_id=p_request
   AND generation=p_generation AND phase='auth_in_flight';
 IF NOT FOUND THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 RETURN true;
END $$;

-- Read-only idempotent confirmation. The trusted server binds these IDs to a
-- verified session and signed checkpoint; this function is not browser auth.
-- No subject hash is needed: retention may scrub it after admission.
CREATE FUNCTION app_private.draft_confirm_complete(p_user uuid,p_session uuid,p_intent uuid)
RETURNS boolean LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $$
BEGIN
 IF p_user IS NULL OR p_session IS NULL OR p_intent IS NULL THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 PERFORM 1 FROM auth.users WHERE id=p_user FOR SHARE;
 PERFORM 1 FROM auth.sessions WHERE id=p_session AND user_id=p_user FOR SHARE;
 PERFORM 1 FROM public.user_moderation WHERE user_id=p_user FOR SHARE;
 IF app_private.signup_pending_identity(p_user,p_session) IS NOT TRUE THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 PERFORM pg_catalog.pg_advisory_xact_lock(pg_catalog.hashtextextended(p_user::text || ':signup',0));
 PERFORM 1 FROM app_private.signup_intents WHERE id=p_intent AND user_id=p_user AND state='consumed' FOR SHARE;
 IF NOT FOUND THEN RETURN false; END IF;
 PERFORM 1 FROM app_private.user_admission WHERE user_id=p_user AND origin='public' AND state='admitted' FOR SHARE;
 IF NOT FOUND THEN RETURN false; END IF;
 PERFORM 1 FROM app_private.signup_completion_fences_draft WHERE user_id=p_user AND owner_session_id=p_session AND intent_id=p_intent AND phase='complete' FOR SHARE;
 IF NOT FOUND THEN RETURN false; END IF;
 RETURN EXISTS(SELECT 1 FROM app_private.onboarding_entries WHERE user_id=p_user AND source_kind='public' AND source_intent=p_intent);
END $$;

-- Input versions are supplied by trusted server; SQL does not independently
-- discover the currently published legal or origin-consent document versions.
CREATE FUNCTION app_private.draft_finish_fenced(p_user uuid,p_session uuid,p_intent uuid,p_subject_hash text,p_request uuid,p_generation uuid,p_current_terms text,p_current_privacy text,p_current_origin text)
RETURNS boolean LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $$
DECLARE f app_private.signup_completion_fences_draft; i app_private.signup_intents;
BEGIN
 -- Only a positively verified completed operation may bypass bound/pending;
 -- stale request/generation must not be reported as a matching finish.
 IF app_private.draft_confirm_complete(p_user,p_session,p_intent) THEN
  RETURN EXISTS(SELECT 1 FROM app_private.signup_completion_fences_draft WHERE user_id=p_user AND owner_session_id=p_session
   AND intent_id=p_intent AND request_id=p_request AND generation=p_generation AND phase='complete');
 END IF;
 PERFORM app_private.draft_completion_context(p_user,p_session,p_intent,p_subject_hash,true,true);
 SELECT * INTO f FROM app_private.signup_completion_fences_draft WHERE user_id=p_user AND intent_id=p_intent
  AND owner_session_id=p_session AND request_id=p_request AND generation=p_generation FOR UPDATE;
 IF NOT FOUND OR f.phase IS DISTINCT FROM 'password_acknowledged' THEN RETURN false; END IF;
 SELECT * INTO STRICT i FROM app_private.signup_intents WHERE id=p_intent;
 IF p_current_terms IS NULL OR p_current_privacy IS NULL OR f.terms_version IS DISTINCT FROM p_current_terms
  OR f.privacy_version IS DISTINCT FROM p_current_privacy OR i.terms_version IS DISTINCT FROM f.terms_version
  OR i.privacy_version IS DISTINCT FROM f.privacy_version THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 IF NOT f.origin_accepted AND i.origin IS NOT NULL THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 IF f.origin_accepted AND NOT f.origin_withdrawn AND NOT f.origin_expired AND i.origin IS NOT NULL AND
  (p_current_origin IS NULL OR f.origin_consent_version IS DISTINCT FROM p_current_origin
   OR i.origin_consent_version IS DISTINCT FROM f.origin_consent_version) THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 -- The proof row may be contended; check its expiry only after acquiring it.
 PERFORM 1 FROM app_private.signup_password_proofs WHERE intent_id=i.id FOR UPDATE;
 IF NOT FOUND THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 UPDATE app_private.signup_password_proofs SET consumed_at=pg_catalog.now()
  WHERE intent_id=i.id AND user_id=p_user AND session_id=p_session AND consumed_at IS NULL AND expires_at>pg_catalog.clock_timestamp();
 IF NOT FOUND THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 -- Recheck after all locks and proof work. A deadline crossed while waiting
 -- must not turn a stale consent into a newly persisted attribution.
 IF i.expires_at IS NULL OR i.expires_at<=pg_catalog.clock_timestamp() THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 IF f.origin_accepted AND NOT f.origin_withdrawn AND NOT f.origin_expired AND i.origin IS NOT NULL THEN
  INSERT INTO app_private.signup_origins(user_id,origin,consent_version,consented_at,expires_at)
   VALUES(p_user,i.origin,i.origin_consent_version,pg_catalog.now(),pg_catalog.now()+interval '12 months');
  PERFORM app_private.append_event(p_user,'signup_origin',p_user,'created','web',NULL,pg_catalog.jsonb_build_object('consent','granted'));
 END IF;
 UPDATE app_private.signup_intents SET state='consumed',consumed_at=pg_catalog.now(),origin=NULL,origin_consent_version=NULL,origin_consented_at=NULL
  WHERE id=i.id AND user_id=p_user AND state='bound';
 IF NOT FOUND THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 UPDATE app_private.user_admission SET origin='public',state='admitted',admitted_at=pg_catalog.now()
  WHERE user_id=p_user AND origin='unclassified' AND state='pending';
 IF NOT FOUND THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 INSERT INTO app_private.onboarding_entries(user_id,source_operation,source_kind,source_intent) VALUES(p_user,NULL,'public',i.id);
 PERFORM app_private.append_event(p_user,'signup_admission',p_user,'created','web',NULL,pg_catalog.jsonb_build_object('user_id',p_user,'origin','public'));
 PERFORM app_private.append_event(p_user,'onboarding_entry',p_user,'created','web',NULL,pg_catalog.jsonb_build_object('user_id',p_user,'eligibility','new_explicit','schema_version',1));
 UPDATE app_private.signup_completion_fences_draft SET phase='complete',updated_at=pg_catalog.now()
  WHERE user_id=p_user AND intent_id=p_intent AND owner_session_id=p_session AND request_id=p_request
   AND generation=p_generation AND phase='password_acknowledged';
 IF NOT FOUND THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 RETURN true;
END $$;

-- Privacy withdrawal is independent of the forced-password flag. It still
-- requires an active verified owner/session and a non-blocked confirmed user.
CREATE FUNCTION app_private.draft_withdraw_identity(p_user uuid,p_session uuid)
RETURNS boolean LANGUAGE sql STABLE SECURITY DEFINER SET search_path='' AS $$
 SELECT p_user IS NOT NULL AND p_session IS NOT NULL AND EXISTS(
  SELECT 1 FROM auth.users u JOIN auth.sessions s ON s.user_id=u.id JOIN public.user_moderation m ON m.user_id=u.id
  WHERE u.id=p_user AND s.id=p_session AND u.email_confirmed_at IS NOT NULL AND NOT u.is_anonymous
   AND u.deleted_at IS NULL AND (u.banned_until IS NULL OR u.banned_until<=pg_catalog.now())
   AND (s.not_after IS NULL OR s.not_after>pg_catalog.now()) AND m.status='active');
$$;
REVOKE ALL ON FUNCTION app_private.draft_withdraw_identity(uuid,uuid) FROM PUBLIC,anon,authenticated,service_role;

-- Monotonic withdrawal across pending and completed states, on the same
-- per-user signup lock as reserve and finish. No new consent is written.
CREATE FUNCTION app_private.draft_withdraw_origin(p_user uuid,p_session uuid,p_intent uuid,p_subject_hash text)
RETURNS boolean LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $$
DECLARE i app_private.signup_intents; a app_private.user_admission; f app_private.signup_completion_fences_draft; removed boolean;
BEGIN
 IF p_user IS NULL OR p_session IS NULL OR p_intent IS NULL THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 PERFORM 1 FROM auth.users WHERE id=p_user FOR SHARE;
 PERFORM 1 FROM auth.sessions WHERE id=p_session AND user_id=p_user FOR SHARE;
 PERFORM 1 FROM public.user_moderation WHERE user_id=p_user FOR SHARE;
 IF app_private.draft_withdraw_identity(p_user,p_session) IS NOT TRUE THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 PERFORM pg_catalog.pg_advisory_xact_lock(pg_catalog.hashtextextended(p_user::text || ':signup',0));
 SELECT * INTO i FROM app_private.signup_intents WHERE id=p_intent AND user_id=p_user FOR UPDATE;
 IF NOT FOUND THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 SELECT * INTO a FROM app_private.user_admission WHERE user_id=p_user FOR UPDATE;
 IF NOT FOUND THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 SELECT * INTO f FROM app_private.signup_completion_fences_draft WHERE user_id=p_user FOR UPDATE;
 IF FOUND AND f.intent_id IS DISTINCT FROM p_intent THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 IF i.state='bound' AND a.origin='unclassified' AND a.state='pending' AND i.subject_hash IS NOT DISTINCT FROM p_subject_hash AND p_subject_hash ~ '^[a-f0-9]{64}$' THEN
  UPDATE app_private.signup_intents SET origin=NULL,origin_consent_version=NULL,origin_consented_at=NULL WHERE id=p_intent;
  UPDATE app_private.signup_completion_fences_draft SET origin_withdrawn=true,updated_at=pg_catalog.now() WHERE user_id=p_user AND intent_id=p_intent;
  RETURN true;
 END IF;
 -- A current verified session of the same owner may withdraw after admission;
 -- unlike idempotent finish, this does not require the original session.
 IF i.state='consumed' AND a.origin='public' AND a.state='admitted' AND f.phase='complete' THEN
  UPDATE app_private.signup_completion_fences_draft SET origin_withdrawn=true,updated_at=pg_catalog.now() WHERE user_id=p_user AND intent_id=p_intent;
  DELETE FROM app_private.signup_origins WHERE user_id=p_user;
  removed:=FOUND;
  IF removed THEN PERFORM app_private.append_event(p_user,'signup_origin',p_user,'deleted','web',NULL,pg_catalog.jsonb_build_object('consent','withdrawn')); END IF;
  RETURN true;
 END IF;
 RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501';
END $$;

-- Separate authenticated settings command: no browser-supplied intent, legal
-- version, checkpoint or password. Runtime derives p_user/p_session from Auth.
CREATE FUNCTION app_private.draft_withdraw_current(p_user uuid,p_session uuid)
RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $$
DECLARE intent_changed integer:=0; stored_changed integer:=0;
BEGIN
 IF p_user IS NULL OR p_session IS NULL THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 PERFORM 1 FROM auth.users WHERE id=p_user FOR SHARE;
 PERFORM 1 FROM auth.sessions WHERE id=p_session AND user_id=p_user FOR SHARE;
 PERFORM 1 FROM public.user_moderation WHERE user_id=p_user FOR SHARE;
 IF app_private.draft_withdraw_identity(p_user,p_session) IS NOT TRUE THEN RAISE EXCEPTION 'Signup unavailable.' USING ERRCODE='42501'; END IF;
 PERFORM pg_catalog.pg_advisory_xact_lock(pg_catalog.hashtextextended(p_user::text || ':signup',0));
 PERFORM 1 FROM app_private.signup_intents WHERE user_id=p_user FOR UPDATE;
 UPDATE app_private.signup_intents SET origin=NULL,origin_consent_version=NULL,origin_consented_at=NULL WHERE user_id=p_user AND state='bound' AND origin IS NOT NULL;
 GET DIAGNOSTICS intent_changed=ROW_COUNT;
 PERFORM 1 FROM app_private.user_admission WHERE user_id=p_user FOR UPDATE;
 PERFORM 1 FROM app_private.signup_completion_fences_draft WHERE user_id=p_user FOR UPDATE;
 UPDATE app_private.signup_completion_fences_draft SET origin_withdrawn=true,updated_at=pg_catalog.now() WHERE user_id=p_user AND NOT origin_withdrawn;
 DELETE FROM app_private.signup_origins WHERE user_id=p_user;
 GET DIAGNOSTICS stored_changed=ROW_COUNT;
 IF intent_changed+stored_changed>0 THEN
  PERFORM app_private.append_event(p_user,'signup_origin',p_user,'deleted','web',NULL,pg_catalog.jsonb_build_object('consent','withdrawn'));
  RETURN pg_catalog.jsonb_build_object('status','withdrawn');
 END IF;
 RETURN pg_catalog.jsonb_build_object('status','absent');
END $$;

-- Hash collisions only over-serialize. Existing writers DO NOT take this lock.
CREATE FUNCTION app_private.draft_feedback_lock(p_user uuid,p_client uuid) RETURNS void
LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $$ BEGIN
 IF p_user IS NULL OR p_client IS NULL THEN RAISE EXCEPTION 'Unavailable.' USING ERRCODE='42501'; END IF;
 PERFORM pg_catalog.pg_advisory_xact_lock(pg_catalog.hashtextextended(p_user::text || ':' || p_client::text || ':feedback',0));
END $$;
REVOKE ALL ON FUNCTION app_private.draft_feedback_lock(uuid,uuid) FROM PUBLIC,anon,authenticated,service_role;

CREATE FUNCTION app_private.draft_feedback_actor(p_user uuid,p_session uuid) RETURNS void
LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $$ BEGIN
 PERFORM 1 FROM auth.users WHERE id=p_user FOR SHARE;
 PERFORM 1 FROM auth.sessions WHERE id=p_session AND user_id=p_user FOR SHARE;
 PERFORM 1 FROM public.user_moderation WHERE user_id=p_user FOR SHARE;
 PERFORM app_private.require_actor(p_user,p_session);
END $$;
REVOKE ALL ON FUNCTION app_private.draft_feedback_actor(uuid,uuid) FROM PUBLIC,anon,authenticated,service_role;

CREATE FUNCTION app_private.draft_feedback_submit(p_user uuid,p_session uuid,p_client uuid,p_category text,p_message text) RETURNS jsonb
LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $$
DECLARE v_barrier app_private.feedback_barriers_draft; v_row app_private.feedback_requests;
BEGIN
 PERFORM app_private.draft_feedback_actor(p_user,p_session);
 IF p_client IS NULL OR p_category IS NULL OR p_category NOT IN ('suggestion','complaint','problem','improvement','other')
  OR p_message IS NULL OR pg_catalog.char_length(pg_catalog.btrim(p_message)) NOT BETWEEN 10 AND 5000
  OR p_message ~ '[\x00-\x08\x0B\x0C\x0E-\x1F\x7F]' THEN RAISE EXCEPTION 'Invalid feedback.' USING ERRCODE='22023'; END IF;
 PERFORM app_private.draft_feedback_lock(p_user,p_client);
 SELECT * INTO v_barrier FROM app_private.feedback_barriers_draft WHERE user_id=p_user AND client_id=p_client FOR UPDATE;
 IF FOUND THEN
  IF v_barrier.phase='cancelled' THEN
   IF EXISTS(SELECT 1 FROM app_private.feedback_requests WHERE user_id=p_user AND client_id=p_client) THEN RAISE EXCEPTION 'Unavailable.' USING ERRCODE='55000'; END IF;
   RETURN pg_catalog.jsonb_build_object('status','cancelled','clientId',p_client);
  END IF;
  SELECT * INTO v_row FROM app_private.feedback_requests WHERE id=v_barrier.feedback_id AND user_id=p_user AND client_id=p_client;
  IF NOT FOUND THEN RAISE EXCEPTION 'Unavailable.' USING ERRCODE='55000'; END IF;
  IF v_row.category IS DISTINCT FROM p_category OR v_row.message IS DISTINCT FROM pg_catalog.btrim(p_message) THEN RAISE EXCEPTION 'Conflicting request.' USING ERRCODE='23505'; END IF;
 ELSE
  -- A committed legacy row may be adopted only under the same keyed lock.
  SELECT * INTO v_row FROM app_private.feedback_requests WHERE user_id=p_user AND client_id=p_client FOR UPDATE;
  IF FOUND THEN
   IF v_row.category IS DISTINCT FROM p_category OR v_row.message IS DISTINCT FROM pg_catalog.btrim(p_message) THEN RAISE EXCEPTION 'Conflicting request.' USING ERRCODE='23505'; END IF;
  ELSE
   INSERT INTO app_private.feedback_requests(user_id,client_id,category,message) VALUES(p_user,p_client,p_category,pg_catalog.btrim(p_message)) RETURNING * INTO v_row;
   PERFORM app_private.append_event(p_user,'feedback_request',v_row.id,'created','web',NULL,
    pg_catalog.jsonb_build_object('id',v_row.id,'category',v_row.category,'status','received'));
  END IF;
  INSERT INTO app_private.feedback_barriers_draft(user_id,client_id,phase,feedback_id) VALUES(p_user,p_client,'submitted',v_row.id);
 END IF;
 RETURN pg_catalog.jsonb_build_object('status','submitted','clientId',p_client,'feedbackId',v_row.id);
END $$;

CREATE FUNCTION app_private.draft_feedback_cancel(p_user uuid,p_session uuid,p_client uuid) RETURNS jsonb
LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $$
DECLARE v_barrier app_private.feedback_barriers_draft; v_legacy_id uuid;
BEGIN
 PERFORM app_private.draft_feedback_actor(p_user,p_session);
 IF p_client IS NULL THEN RAISE EXCEPTION 'Unavailable.' USING ERRCODE='42501'; END IF;
 PERFORM app_private.draft_feedback_lock(p_user,p_client);
 SELECT * INTO v_barrier FROM app_private.feedback_barriers_draft WHERE user_id=p_user AND client_id=p_client FOR UPDATE;
 IF NOT FOUND THEN
  -- Only a committed same-owner legacy row proves an earlier submission.
  SELECT id INTO v_legacy_id FROM app_private.feedback_requests WHERE user_id=p_user AND client_id=p_client FOR UPDATE;
  IF FOUND THEN
   INSERT INTO app_private.feedback_barriers_draft(user_id,client_id,phase,feedback_id) VALUES(p_user,p_client,'submitted',v_legacy_id);
   RETURN pg_catalog.jsonb_build_object('status','submitted','clientId',p_client,'feedbackId',v_legacy_id);
  END IF;
  INSERT INTO app_private.feedback_barriers_draft(user_id,client_id,phase) VALUES(p_user,p_client,'cancelled');
  RETURN pg_catalog.jsonb_build_object('status','cancelled','clientId',p_client);
 END IF;
 IF v_barrier.phase='cancelled' THEN
   IF EXISTS(SELECT 1 FROM app_private.feedback_requests WHERE user_id=p_user AND client_id=p_client) THEN RAISE EXCEPTION 'Unavailable.' USING ERRCODE='55000'; END IF;
   RETURN pg_catalog.jsonb_build_object('status','cancelled','clientId',p_client);
  END IF;
 IF NOT EXISTS(SELECT 1 FROM app_private.feedback_requests WHERE id=v_barrier.feedback_id AND user_id=p_user AND client_id=p_client) THEN RAISE EXCEPTION 'Unavailable.' USING ERRCODE='55000'; END IF;
 RETURN pg_catalog.jsonb_build_object('status','submitted','clientId',p_client,'feedbackId',v_barrier.feedback_id);
END $$;

CREATE FUNCTION app_private.draft_feedback_status(p_user uuid,p_session uuid,p_client uuid) RETURNS jsonb

BEGIN
 PERFORM app_private.draft_feedback_actor(p_user,p_session);

 -- Even the status path locks before adopting a committed legacy row.
 PERFORM app_private.draft_feedback_lock(p_user,p_client);
 SELECT * INTO v_barrier FROM app_private.feedback_barriers_draft WHERE user_id=p_user AND client_id=p_client FOR UPDATE;
 IF NOT FOUND THEN
  SELECT id INTO v_legacy_id FROM app_private.feedback_requests WHERE user_id=p_user AND client_id=p_client FOR UPDATE;
  IF FOUND THEN
   INSERT INTO app_private.feedback_barriers_draft(user_id,client_id,phase,feedback_id) VALUES(p_user,p_client,'submitted',v_legacy_id);
   RETURN pg_catalog.jsonb_build_object('status','submitted','clientId',p_client,'feedbackId',v_legacy_id);
  END IF;
  RETURN pg_catalog.jsonb_build_object('status','unknown','clientId',p_client);
 END IF;
 IF v_barrier.phase='cancelled' THEN
   IF EXISTS(SELECT 1 FROM app_private.feedback_requests WHERE user_id=p_user AND client_id=p_client) THEN RAISE EXCEPTION 'Unavailable.' USING ERRCODE='55000'; END IF;
   RETURN pg_catalog.jsonb_build_object('status','cancelled','clientId',p_client);
  END IF;
 IF NOT EXISTS(SELECT 1 FROM app_private.feedback_requests WHERE id=v_barrier.feedback_id AND user_id=p_user AND client_id=p_client) THEN RAISE EXCEPTION 'Unavailable.' USING ERRCODE='55000'; END IF;
 RETURN pg_catalog.jsonb_build_object('status','submitted','clientId',p_client,'feedbackId',v_barrier.feedback_id);
END $$;

create function public.signup_metrics(p_actor uuid,p_session uuid,p_start date,p_end date,p_zone text,p_bucket text)
returns jsonb language plpgsql security definer set search_path='' as $$
declare result jsonb;
begin
 perform app_private.admin_require_master(p_actor,p_session);
 if p_start is null or p_end is null or p_end<p_start or p_end-p_start>365
   or p_zone not in ('America/Fortaleza','UTC') or p_zone is null
   or p_bucket not in ('day','week','month') or p_bucket is null then
  raise exception 'Invalid reporting period.' using errcode='22023';
 end if;
 with population as materialized (
  select a.admitted_at, case when o.expires_at>now() then o.origin else null end as origin
  from app_private.user_admission a
  left join app_private.signup_origins o on o.user_id=a.user_id
  where a.state='admitted' and a.origin='public'
   and a.admitted_at >= (p_start::timestamp at time zone p_zone)
   and a.admitted_at < ((p_end+1)::timestamp at time zone p_zone)
 ), totals as (select count(*)::int as n from population),
 counts as (
  select date_trunc(p_bucket, admitted_at at time zone p_zone)::date as period, count(*)::int as n
  from population group by 1
 ), periods as (
  select generate_series(date_trunc(p_bucket,p_start::timestamp),date_trunc(p_bucket,p_end::timestamp),
   case p_bucket when 'day' then interval '1 day' when 'week' then interval '1 week' else interval '1 month' end)::date as period
 ), dimensions(name) as (values ('utm_source'),('utm_medium'),('utm_campaign'),('utm_term'),('utm_content')),
 classified as materialized (
  select d.name,case when p.origin->>d.name is null then null
   when d.name='utm_source' and lower(btrim(p.origin->>d.name) collate "C")='google' then 'google'
   when d.name='utm_source' and lower(btrim(p.origin->>d.name) collate "C")='instagram' then 'instagram'
   when d.name='utm_medium' and lower(btrim(p.origin->>d.name) collate "C")='organic' then 'organic'
   when d.name='utm_medium' and lower(btrim(p.origin->>d.name) collate "C")='email' then 'email'
   else '__other__' end as value
  from dimensions d cross join population p
 ), grouped as (
  select name,value,count(*)::int as n from classified group by name,value
 ), normalized as (
  select name,case when value is null then 'not-informed' when value='__other__' or n<5 then 'other' else 'value' end as kind,
   case when value is null or value='__other__' or n<5 then null else value end as value,n from grouped
 ), merged as (select name,kind,value,sum(n)::int as n from normalized group by name,kind,value),
 distributions as (
  select d.name,coalesce(jsonb_agg(jsonb_build_object('kind',m.kind,'value',m.value,'count',m.n)
   order by m.n desc,m.kind,m.value) filter(where m.kind is not null),'[]'::jsonb) as entries
  from dimensions d left join merged m on m.name=d.name group by d.name
 )
 select jsonb_build_object('dictionaryVersion','sha256:b27588bd3279e952f29b3da900351e662a06bca55a565c2c9118c70c9946e6ec','totalPublic',t.n,'start',p_start,'end',p_end,'zone',p_zone,'bucket',p_bucket,
  'series',(select coalesce(jsonb_agg(jsonb_build_object('period',p.period,'count',coalesce(c.n,0)) order by p.period),'[]'::jsonb) from periods p left join counts c using(period)),
  'distributions',(select jsonb_object_agg(name,entries) from distributions),
  'stock',(select jsonb_build_object('total',count(*)::int,'legacy',count(*) filter(where a.origin='legacy')::int,
   'admin',count(*) filter(where a.origin='admin')::int,'public',count(*) filter(where a.origin='public')::int,
   'unclassified',count(*) filter(where a.origin is null or a.origin='unclassified')::int)
   from public.profiles p join auth.users u on u.id=p.user_id and u.deleted_at is null and not u.is_anonymous
   left join app_private.user_admission a on a.user_id=p.user_id and a.state='admitted'))
 into result from totals t;
 return result;
end $$;
