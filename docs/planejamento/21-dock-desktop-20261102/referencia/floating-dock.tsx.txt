"use client"

import * as React from "react"
import {
  AnimatePresence,
  MotionConfig,
  motion,
  useMotionValue,
  useReducedMotion,
  useSpring,
  useTransform,
  type MotionValue,
} from "motion/react"

/**
 * Floating Dock — uma barra de ícones flutuante, como a dock do macOS: o ícone sob o mouse cresce, os vizinhos crescem
 * menos conforme a distância, e o nome dele aparece num balão por cima.
 *
 * Na tela larga (768 px ou mais) é a dock. Na estreita, um botão redondo que abre os ícones numa coluna, para cima.
 *
 * - Cada item é um link (com href) ou um botão (com onClick). Com active, ganha um ponto embaixo, como o app aberto
 *   na dock do macOS, e o leitor de tela o lê como a página atual.
 * - O crescer segue o mouse e a caneta; no toque, nada fica preso grande. Pelo teclado, o foco mostra o balão.
 * - As cores vêm de variáveis CSS (--fd-bg, --fd-item, --fd-icon…), com um tema claro e um escuro (theme).
 * - Funciona dentro de um elemento com transform: scale (um zoom, uma prévia encolhida): a distância desconta a escala.
 *
 * Posição, margem e largura aceitam as classes do projeto (Tailwind, CSS Modules, CSS comum). Os ícones do botão são
 * os do Tabler (licença MIT), desenhados aqui.
 *
 * O React e o Motion são os únicos imports. Sem Tailwind, sem arquivo CSS.
 */

export type FloatingDockItem = {
  /** O nome do item: vai no balão e é o que o leitor de tela lê. */
  title: string
  /** Um <svg> ou um <img>: ele ocupa o espaço do ícone inteiro. */
  icon: React.ReactNode
  /** Com href, o item é um link. */
  href?: string
  /** O clique no item, com ou sem href. */
  onClick?: React.MouseEventHandler<HTMLElement>
  /** O item da página atual: um ponto embaixo dele. */
  active?: boolean
  separatorBefore?: boolean
}

export type FloatingDockProps = {
  items: FloatingDockItem[]
  /** A classe da dock, na tela larga. */
  desktopClassName?: string
  /** A classe do botão e da coluna, na tela estreita. */
  mobileClassName?: string
  /** O tema das cores. */
  theme?: "light" | "dark"
  /** O nome da navegação para o leitor de tela. */
  label?: string
}

// O React 19 sobe uma <style> com href e precedence para o <head>, uma vez só para todas as docks. O React 18 não conhece
// isso: lá ela vai dentro da dock, escondida e com o CSS cru, porque como filho de texto o SSR do 18 escaparia as aspas
// e quebraria os seletores.
const HOISTS_STYLES = Number(React.version.split(".")[0]) >= 19

function FloatingDockStyles() {
  if (HOISTS_STYLES) {
    const hoist = { href: "floating-dock", precedence: "default" } as React.StyleHTMLAttributes<HTMLStyleElement>
    return <style {...hoist}>{FD_CSS}</style>
  }
  return <style hidden dangerouslySetInnerHTML={{ __html: FD_CSS }} />
}

export function FloatingDock({ items, desktopClassName, mobileClassName, theme = "light", label = "Navegação" }: FloatingDockProps) {
  return (
    <MotionConfig reducedMotion="user">
      <FloatingDockStyles />
      <FloatingDockDesktop items={items} className={desktopClassName} theme={theme} label={label} />
      <FloatingDockMobile items={items} className={mobileClassName} theme={theme} label={label} />
    </MotionConfig>
  )
}

type PartProps = { items: FloatingDockItem[]; className?: string; theme: "light" | "dark"; label: string }

function cx(...names: (string | false | undefined)[]) {
  return names.filter(Boolean).join(" ")
}

/** O link ou o botão de um item. */
function ItemControl({
  item,
  className,
  children,
  onFocus,
  onBlur,
  onClick,
}: {
  item: FloatingDockItem
  className: string
  children: React.ReactNode
  onFocus?: React.FocusEventHandler<HTMLElement>
  onBlur?: React.FocusEventHandler<HTMLElement>
  onClick?: React.MouseEventHandler<HTMLElement>
}) {
  const common = {
    className: cx(className, item.separatorBefore && "fd-group-start"),
    "aria-label": item.title,
    "aria-current": item.active ? ("page" as const) : undefined,
    onFocus,
    onBlur,
    onClick: (e: React.MouseEvent<HTMLElement>) => {
      item.onClick?.(e)
      onClick?.(e)
    },
  }
  return item.href != null ? (
    <a href={item.href} {...common}>
      {children}
    </a>
  ) : (
    <button type="button" {...common}>
      {children}
    </button>
  )
}

function FloatingDockMobile({ items, className, theme, label }: PartProps) {
  const [open, setOpen] = React.useState(false)
  const root = React.useRef<HTMLDivElement>(null)
  const toggle = React.useRef<HTMLButtonElement>(null)
  const listId = React.useId()

  // aberta, fecha com o Esc (o foco volta ao botão) e com um toque fora
  React.useEffect(() => {
    if (!open) return
    const onKey = (e: KeyboardEvent) => {
      if (e.key !== "Escape" || e.defaultPrevented) return
      e.preventDefault()
      setOpen(false)
      toggle.current?.focus()
    }
    const onDown = (e: PointerEvent) => {
      if (!root.current?.contains(e.target as Node)) setOpen(false)
    }
    document.addEventListener("keydown", onKey)
    document.addEventListener("pointerdown", onDown)
    return () => {
      document.removeEventListener("keydown", onKey)
      document.removeEventListener("pointerdown", onDown)
    }
  }, [open])

  return (
    <nav ref={root} className={cx("fd-mobile", className)} data-fd-theme={theme} aria-label={label}>
      <AnimatePresence>
        {open && (
          <motion.div id={listId} className="fd-mobile__list">
            {items.map((item, idx) => (
              <motion.div
                key={item.title}
                initial={{ opacity: 0, y: 10 }}
                animate={{ opacity: 1, y: 0 }}
                exit={{ opacity: 0, transition: { duration: 0.1 } }}
                transition={{ duration: 0.18, delay: idx * 0.012 }}
              >
                <ItemControl item={item} className="fd-mobile__item" onClick={() => setOpen(false)}>
                  <span className="fd-icon fd-mobile__icon">{item.icon}</span>
                  <span className="fd-mobile__label">{item.title}</span>
                </ItemControl>
              </motion.div>
            ))}
          </motion.div>
        )}
      </AnimatePresence>
      <button
        ref={toggle}
        type="button"
        className="fd-mobile__toggle"
        onClick={() => setOpen(o => !o)}
        aria-expanded={open}
        aria-controls={open ? listId : undefined}
        aria-label={open ? "Fechar o menu" : "Abrir o menu"}
      >
        <IconLayoutNavbarCollapse />
      </button>
    </nav>
  )
}

function FloatingDockDesktop({ items, className, theme, label }: PartProps) {
  const mouseX = useMotionValue(Infinity)
  const reducedMotion = useReducedMotion()
  // só o mouse e a caneta: no toque o ponteiro some depois do toque, e o ícone ficaria preso grande
  const follow = (e: React.PointerEvent) => {
    if (e.pointerType !== "touch" && !reducedMotion) mouseX.set(e.clientX)
  }
  return (
    <nav
      className={cx("fd-dock", className)}
      data-fd-theme={theme}
      aria-label={label}
      onPointerMove={follow}
      onPointerLeave={() => mouseX.set(Infinity)}
    >
      {items.map(item => (
        <IconContainer mouseX={mouseX} key={item.title} item={item} />
      ))}
    </nav>
  )
}

const SPRING = { mass: 0.16, stiffness: 220, damping: 20 }

function IconContainer({ mouseX, item }: { mouseX: MotionValue<number>; item: FloatingDockItem }) {
  const ref = React.useRef<HTMLDivElement>(null)

  // a distância do mouse ao centro do ícone, em px da dock: dentro de um transform: scale a tela mede encolhido, e a
  // escala sai da largura medida dividida pela de layout
  const distance = useTransform(mouseX, val => {
    const el = ref.current
    if (!el) return Infinity
    const bounds = el.getBoundingClientRect()
    const scale = el.offsetWidth > 0 ? bounds.width / el.offsetWidth : 1
    return (val - bounds.left - bounds.width / 2) / scale
  })

  const width = useSpring(useTransform(distance, [-110, 0, 110], [44, 64, 44]), SPRING)
  const height = useSpring(useTransform(distance, [-110, 0, 110], [44, 64, 44]), SPRING)
  const widthIcon = useSpring(useTransform(distance, [-110, 0, 110], [22, 30, 22]), SPRING)
  const heightIcon = useSpring(useTransform(distance, [-110, 0, 110], [22, 30, 22]), SPRING)

  const [hovered, setHovered] = React.useState(false)
  const [focused, setFocused] = React.useState(false)

  return (
    <ItemControl
      item={item}
      className="fd-dock__item"
      onFocus={e => setFocused(e.currentTarget.matches(":focus-visible"))}
      onBlur={() => setFocused(false)}
    >
      <motion.div
        ref={ref}
        style={{ width, height }}
        onPointerEnter={e => e.pointerType !== "touch" && setHovered(true)}
        onPointerLeave={() => setHovered(false)}
        className="fd-dock__tile"
      >
        <AnimatePresence>
          {(hovered || focused) && (
            <motion.div
              initial={{ opacity: 0, y: 10, x: "-50%" }}
              animate={{ opacity: 1, y: 0, x: "-50%" }}
              exit={{ opacity: 0, y: 2, x: "-50%" }}
              className="fd-dock__tip"
              aria-hidden="true"
            >
              {item.title}
            </motion.div>
          )}
        </AnimatePresence>
        <motion.div style={{ width: widthIcon, height: heightIcon }} className="fd-icon">
          {item.icon}
        </motion.div>
      </motion.div>
    </ItemControl>
  )
}

function IconLayoutNavbarCollapse() {
  return (
    <svg
      viewBox="0 0 24 24"
      fill="none"
      stroke="currentColor"
      strokeWidth="2"
      strokeLinecap="round"
      strokeLinejoin="round"
      aria-hidden="true"
      focusable="false"
    >
      <path d="M4 18v-12a2 2 0 0 1 2 -2h12a2 2 0 0 1 2 2v12a2 2 0 0 1 -2 2h-12a2 2 0 0 1 -2 -2" />
      <path d="M4 9h16" />
      <path d="M10 16l2 -2l2 2" />
    </svg>
  )
}
